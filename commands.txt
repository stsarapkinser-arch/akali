# ==========================================================
# БАЗА ЗНАНИЙ АССИСТЕНТА "АКАЛИ" (Kali Linux + KDE Plasma 6)
# Формат: bash_команда -> фраза_1, фраза_2, фраза_3
#
# ПРАВИЛА ЗАПОЛНЕНИЯ:
#   1. GUI-приложение — всегда `&` в конце.
#   2. TUI-приложение — через `konsole -e <cmd> &`.
#   3. xdotool НЕ работает на Wayland — только qdbus/qdbus6.
#   4. Управление питанием только через systemctl.
#   5. Команды с pipe (|) не поддерживаются в голосовом вводе.
# ==========================================================

qdbus org.kde.kglobalaccel /component/kwin invokeShortcut "Window Quick Tile Left" & -> прижми окно влево, окно слева, тайл влево
qdbus org.kde.kglobalaccel /component/kwin invokeShortcut "Window Quick Tile Right" & -> прижми окно вправо, окно справа, тайл вправо

# === СЕССИЯ ===
# ВНИМАНИЕ: команды poweroff/reboot/suspend/hibernate сюда НЕ добавляются.
# Они обрабатываются модулем akali/core/power_guard.py с exact-match,
# чтобы случайный шум или галлюцинация LLM не выключили систему.
loginctl lock-session -> заблокируй экран, заблокируй компьютер, лок скрин
clear -> очисти терминал, убери текст, очисти экран

# === ЗВУК (PipeWire-pulse / PulseAudio) ===
pactl set-sink-volume @DEFAULT_SINK@ +10% -> громче, увеличь громкость, прибавь звук
pactl set-sink-volume @DEFAULT_SINK@ -10% -> тише, уменьши громкость, убавь звук
pactl set-sink-volume @DEFAULT_SINK@ 50% -> установи громкость пятьдесят, средняя громкость
pactl set-sink-volume @DEFAULT_SINK@ 100% -> максимальная громкость, на полную, громкость сто
pactl set-sink-mute @DEFAULT_SINK@ toggle -> отключи звук, включи звук, мьют, без звука
pactl list short sinks -> список устройств вывода, какие колонки, аудио устройства
pactl info -> информация о звуке, аудио сервер

# === МЕДИАПЛЕЕР (playerctl) ===

playerctl previous -> предыдущий трек, назад песня, прошлый трек
playerctl stop -> стоп музыка, останови плеер
playerctl metadata -> что играет, какой трек сейчас, инфо о треке

# === ФАЙЛЫ И ДИСКИ ===

du -sh ~ -> сколько весит домашняя, размер home, занятое место в домашней

# === ПРОЦЕССЫ И СЕРВИСЫ ===
konsole -e htop & -> открой htop, мониторинг процессов в терминале
konsole -e btop & -> открой btop, красивый монитор
ps aux --sort=-%mem -> топ по памяти, что жрёт память, прожорливые процессы
ps aux --sort=-%cpu -> топ по процессору, что нагружает процессор
systemctl list-units --type=service --state=running --no-pager -> запущенные сервисы, что работает в фоне
systemctl --failed --no-pager -> упавшие сервисы, ошибки сервисов
journalctl -p err -n 30 --no-pager -> последние ошибки, ошибки в журнале
dmesg --level=err,warn -> сообщения ядра, ошибки ядра

# === МОНИТОРИНГ РЕСУРСОВ ===

# === СЕТЬ: ДИАГНОСТИКА ===

systemctl restart NetworkManager -> перезагрузи сеть, перезапусти вайфай, сбрось интернет
